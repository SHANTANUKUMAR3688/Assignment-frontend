// tuple
type Product = readonly[number, string, number]

const product1: Product = [1, 'Laptop', 75000]

const showProduct = (p: Product) => {
  const [id, name, price] = p
  console.log(id, name, price)
}

showProduct(product1)

// union
function display(value: string | number) {
  if (typeof value === "string") {
    console.log("String:", value.toUpperCase());
  } else {
    console.log("Number:", value + 10);
  }
}

display("hello");
display(20);

// Intersection
type Person = {
  name: string
}

type Employee = {
  id: number
}

type Staff = Person & Employee

const staff1: Staff = {
  name: "Shantanu",
  id: 101
}

console.log(staff1)


type A = {
  a: number
}

type B = {
  b: number
}

function printValue(obj: A & B) {
  console.log(obj.a + obj.b)
}

printValue({ a: 5, b: 10 })


// generics
function identity<T>(value: T): T {
  return value;
}

console.log(identity<string>("Hello"));
console.log(identity<number>(10));


function getFirstElement<T>(arr: T[]): T {
  return arr[0];
}

console.log(getFirstElement<number>([10, 20, 30]));
console.log(getFirstElement<string>(["a", "b", "c"]));


interface Box<T> {
  value: T;
}

const numberBox: Box<number> = { value: 100 };
const stringBox: Box<string> = { value: "Hello" };

console.log(numberBox.value);
console.log(stringBox.value);
